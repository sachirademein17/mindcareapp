🗄️ Step 1: RDS PostgreSQL Database Setup

1.1 Create RDS Security Group

# Create security group for RDS
aws ec2 create-security-group \
    --group-name mindcare-rds-sg \
    --description "Security group for MindCare RDS PostgreSQL"

# Add PostgreSQL access rule
aws ec2 authorize-security-group-ingress \
    --group-name mindcare-rds-sg \
    --protocol tcp \
    --port 5432 \
    --cidr 10.0.0.0/16

1.2 Create RDS PostgreSQL Instance

aws rds create-db-instance \
    --db-instance-identifier mindcare-db \
    --db-instance-class db.t3.micro \
    --engine postgres \
    --engine-version 15.4 \
    --master-username postgres \
    --master-user-password "YourSecurePassword123!" \
    --allocated-storage 20 \
    --storage-type gp2 \
    --backup-retention-period 7 \
    --storage-encrypted \
    --db-name mindcare_production
⚠️ Important: RDS instance creation takes 5-10 minutes. Save your password securely!

1.3 Get RDS Endpoint

# Get the endpoint once RDS is available
aws rds describe-db-instances \
    --db-instance-identifier mindcare-db \
    --query 'DBInstances[0].Endpoint.Address' \
    --output text

Save this endpoint - you'll need it for your backend configuration.

📁 Step 2: S3 Bucket Configuration

2.1 Create S3 Buckets

# Bucket for file uploads (CVs, documents)
aws s3 mb s3://mindcare-file-uploads-yourname --region us-east-1

# Bucket for frontend static files
aws s3 mb s3://mindcare-frontend-yourname --region us-east-1
⚠️ Important: S3 bucket names must be globally unique. Replace "yourname" with something unique.

2.2 Configure File Uploads Bucket CORS

Create a file named cors-config.json:

{
    "CORSRules": [
        {
            "AllowedHeaders": ["*"],
            "AllowedMethods": ["GET", "PUT", "POST", "DELETE"],
            "AllowedOrigins": ["*"],
            "ExposeHeaders": ["ETag"],
            "MaxAgeSeconds": 3000
        }
    ]
}
                    
# Apply CORS configuration
aws s3api put-bucket-cors \
    --bucket mindcare-file-uploads-yourname \
    --cors-configuration file://cors-config.json

2.3 Configure Frontend Bucket for Static Hosting

# Enable static website hosting
aws s3 website s3://mindcare-frontend-yourname \
    --index-document index.html \
    --error-document index.html

2.4 Create IAM User for S3 Access

# Create IAM user for application S3 access
aws iam create-user --user-name mindcare-s3-user

# Create access key
aws iam create-access-key --user-name mindcare-s3-user

# Attach S3 policy
aws iam attach-user-policy \
    --user-name mindcare-s3-user \
    --policy-arn arn:aws:iam::aws:policy/AmazonS3FullAccess
⚠️ Important: Save the Access Key ID and Secret Access Key for your backend configuration.
🖥️ Step 3: EC2 Ubuntu Server Setup

3.1 Create EC2 Key Pair

# Create key pair
aws ec2 create-key-pair \
    --key-name mindcare-key \
    --query 'KeyMaterial' \
    --output text > mindcare-key.pem

# Set correct permissions (Linux/Mac)
chmod 400 mindcare-key.pem

3.2 Create Security Group for EC2

# Create security group
aws ec2 create-security-group \
    --group-name mindcare-ec2-sg \
    --description "Security group for MindCare EC2 instance"

# Add access rules
aws ec2 authorize-security-group-ingress --group-name mindcare-ec2-sg --protocol tcp --port 22 --cidr 0.0.0.0/0
aws ec2 authorize-security-group-ingress --group-name mindcare-ec2-sg --protocol tcp --port 80 --cidr 0.0.0.0/0
aws ec2 authorize-security-group-ingress --group-name mindcare-ec2-sg --protocol tcp --port 443 --cidr 0.0.0.0/0
aws ec2 authorize-security-group-ingress --group-name mindcare-ec2-sg --protocol tcp --port 5000 --cidr 0.0.0.0/0

3.3 Launch EC2 Ubuntu Instance

# Find latest Ubuntu 22.04 LTS AMI
aws ec2 describe-images \
    --owners 099720109477 \
    --filters "Name=name,Values=ubuntu/images/hvm-ssd/ubuntu-jammy-22.04-amd64-server-*" \
    --query 'Images | sort_by(@, &CreationDate) | [-1].ImageId' \
    --output text

# Launch instance (replace ami-id with result from above)
aws ec2 run-instances \
    --image-id ami-0c02fb55956c7d316 \
    --count 1 \
    --instance-type t3.medium \
    --key-name mindcare-key \
    --security-groups mindcare-ec2-sg \
    --tag-specifications 'ResourceType=instance,Tags=[{Key=Name,Value=MindCare-Backend}]'

3.4 Get EC2 Instance Public IP

# Get public IP address
aws ec2 describe-instances \
    --filters "Name=tag:Name,Values=MindCare-Backend" \
    --query 'Reservations[0].Instances[0].PublicIpAddress' \
    --output text

Save this IP address - you'll use it to connect to your server.

3.5 Connect and Install Dependencies

# Connect via SSH
ssh -i mindcare-key.pem ubuntu@YOUR_EC2_PUBLIC_IP

# Update system
sudo apt update && sudo apt upgrade -y

# Install Node.js 18
curl -fsSL https://deb.nodesource.com/setup_18.x | sudo -E bash -
sudo apt-get install -y nodejs

# Install additional tools
sudo npm install -g pm2
sudo apt install -y nginx postgresql-client git unzip

# Create application directory
sudo mkdir -p /var/www/mindcare
sudo chown ubuntu:ubuntu /var/www/mindcare
🚀 Step 4: Backend Deployment

4.1 Upload Backend Code

On your local machine:

# Go to your project directory
cd backend
npm run build

# Create deployment package
tar -czf mindcare-backend.tar.gz dist/ package.json package-lock.json src/ uploads/

# Upload to EC2 (replace with your IP)
scp -i mindcare-key.pem mindcare-backend.tar.gz ubuntu@YOUR_EC2_IP:/var/www/mindcare/

4.2 Set Up Backend on Server

# SSH into your server
ssh -i mindcare-key.pem ubuntu@YOUR_EC2_IP

# Navigate and extract
cd /var/www/mindcare
tar -xzf mindcare-backend.tar.gz

# Install dependencies
npm ci --production

# Create uploads directory
mkdir -p uploads
chmod 755 uploads

4.3 Configure Environment Variables

# Create production environment file
nano .env

Add these environment variables (update with your actual values):

NODE_ENV=production
PORT=5000

# Database Configuration
DB_HOST=your-rds-endpoint.region.rds.amazonaws.com
DB_PORT=5432
DB_NAME=mindcare_production
DB_USER=postgres
DB_PASSWORD=YourSecurePassword123!

# JWT Configuration
JWT_SECRET=your-super-secure-jwt-secret-for-production-min-32-chars

# AWS S3 Configuration
AWS_REGION=us-east-1
AWS_ACCESS_KEY_ID=your-s3-access-key-id
AWS_SECRET_ACCESS_KEY=your-s3-secret-access-key
AWS_S3_BUCKET=mindcare-file-uploads-yourname

# CORS Configuration
CORS_ORIGIN=https://your-domain.com

# File Upload Configuration
MAX_FILE_SIZE=10485760
UPLOAD_PATH=./uploads

4.4 Test Database Connection

# Test PostgreSQL connection
PGPASSWORD=YourSecurePassword123! psql \
    -h your-rds-endpoint.region.rds.amazonaws.com \
    -U postgres \
    -d mindcare_production \
    -c "SELECT version();"

4.5 Start Application with PM2

# Run database migrations if needed
npm run migrate

# Start the application
pm2 start dist/app.js --name mindcare-backend

# Save PM2 configuration
pm2 save
pm2 startup systemd

# Check status
pm2 status
pm2 logs mindcare-backend

4.6 Configure Nginx

# Create Nginx configuration
sudo nano /etc/nginx/sites-available/mindcare

Add this configuration:

server {
    listen 80;
    server_name your-domain.com www.your-domain.com;

    location / {
        proxy_pass http://localhost:5000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;
    }
}
                    
# Enable the site
sudo ln -s /etc/nginx/sites-available/mindcare /etc/nginx/sites-enabled/
sudo rm /etc/nginx/sites-enabled/default
sudo nginx -t
sudo systemctl reload nginx
🌐 Step 5: Frontend Deployment

5.1 Build and Deploy Frontend

On your local machine:

# Go to frontend directory
cd frontend

# Create production environment
echo "VITE_API_URL=https://your-domain.com" > .env.production
echo "VITE_NODE_ENV=production" >> .env.production

# Build for production
npm ci
npm run build

# Upload to S3 bucket
aws s3 sync dist/ s3://mindcare-frontend-yourname --delete

# Set proper content types
aws s3 cp s3://mindcare-frontend-yourname/index.html s3://mindcare-frontend-yourname/index.html \
    --metadata-directive REPLACE \
    --content-type "text/html" \
    --cache-control "no-cache"

5.2 Make Frontend Bucket Public

Create bucket-policy.json:

{
    "Version": "2012-10-17",
    "Statement": [
        {
            "Sid": "PublicReadGetObject",
            "Effect": "Allow",
            "Principal": "*",
            "Action": "s3:GetObject",
            "Resource": "arn:aws:s3:::mindcare-frontend-yourname/*"
        }
    ]
}
                    
# Apply bucket policy
aws s3api put-bucket-policy --bucket mindcare-frontend-yourname --policy file://bucket-policy.json

💰 Cost Breakdown

AWS Free Tier (First 12 months)

Service Free Tier Limits After Free Tier (Monthly)
EC2 t2.micro 750 hours/month (FREE) ~$8-10
EC2 t3.medium Not included in free tier ~$30-40
RDS db.t3.micro 750 hours/month (FREE) ~$15-20
S3 Storage 5 GB (FREE) $0.023/GB
Data Transfer 15 GB outbound (FREE) $0.09/GB

💡 Cost Optimization Tips:

  • Start with t2.micro for testing (free tier eligible)
  • Use CloudWatch to monitor resource usage
  • Set up billing alerts
  • Stop instances when not in use
  • Use S3 lifecycle policies for old files

Estimated Monthly Costs (After Free Tier)

  • Light Usage (t2.micro): ~$28/month
  • Medium Usage (t3.medium): ~$70/month
  • Production Usage (t3.large): ~$130/month
